cmake_minimum_required(VERSION 3.18)
project(QES VERSION 2.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# PROJECT_IS_TOP_LEVEL was introduced in CMake 3.21.
if(CMAKE_VERSION VERSION_LESS 3.21)
  if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(PROJECT_IS_TOP_LEVEL ON)
  else()
    set(PROJECT_IS_TOP_LEVEL OFF)
  endif()
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Modules")
set(QES_VERSION_INFO "v${PROJECT_VERSION}")
option(QES_ENABLE_IO "Build file readers/writers and their dependencies" ON)
option(QES_ENABLE_CUDA "Build CUDA acceleration" OFF)
option(QES_ENABLE_OPTIX "Build OptiX acceleration (requires CUDA)" OFF)
set(QES_STANDALONE_DEFAULT OFF)
if(PROJECT_IS_TOP_LEVEL AND QES_ENABLE_IO)
  set(QES_STANDALONE_DEFAULT ON)
endif()
option(QES_BUILD_EXECUTABLES "Build standalone executables" ${QES_STANDALONE_DEFAULT})
option(QES_BUILD_EXAMPLES "Build examples" ${QES_STANDALONE_DEFAULT})
option(QES_BUILD_PLUME "Build QES-Plume" ${QES_STANDALONE_DEFAULT})
option(QES_BUILD_FIRE "Build QES-Fire" ${QES_STANDALONE_DEFAULT})
option(QES_BUILD_DOCUMENTATION "Build API documentation" OFF)
option(ENABLE_OPENMP "Enable multithreaded support" OFF)
option(ENABLE_DEV_MODE "Enable development warnings" OFF)
option(ENABLE_TESTS "Enable the standalone test suite" OFF)
option(ENABLE_CLANG_TIDY "Enable clang-tidy" OFF)
option(ENABLE_CPPCHECK "Enable cppcheck" OFF)
if(NOT QES_ENABLE_IO AND (QES_BUILD_EXECUTABLES OR QES_BUILD_EXAMPLES OR QES_BUILD_PLUME OR QES_BUILD_FIRE))
  message(FATAL_ERROR "Executables, examples, plume and fire require QES_ENABLE_IO=ON")
endif()
if(QES_BUILD_FIRE AND NOT QES_BUILD_PLUME)
  message(FATAL_ERROR "QES-Fire requires QES_BUILD_PLUME=ON")
endif()
if(PROJECT_IS_TOP_LEVEL AND NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
endif()

# Only header-only Boost is needed by the numeric core (DateTime >=1.73).
find_package(Boost 1.73 REQUIRED)
add_library(qesbuild INTERFACE)
target_compile_features(qesbuild INTERFACE cxx_std_17)
target_compile_definitions(qesbuild INTERFACE QES_ENABLE_IO=$<BOOL:${QES_ENABLE_IO}> QES_VERSION_INFO="${QES_VERSION_INFO}")
target_include_directories(qesbuild INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(qesbuild INTERFACE Boost::headers)
if(ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  target_link_libraries(qesbuild INTERFACE OpenMP::OpenMP_CXX)
endif()
if(ENABLE_DEV_MODE)
  target_compile_options(qesbuild INTERFACE "$<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-pedantic>")
endif()
if(ENABLE_CLANG_TIDY)
  find_program(QES_CLANG_TIDY clang-tidy REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${QES_CLANG_TIDY}")
endif()
if(ENABLE_CPPCHECK)
  find_program(QES_CPPCHECK cppcheck REQUIRED)
  set(CMAKE_CXX_CPPCHECK "${QES_CPPCHECK}")
endif()

if(QES_ENABLE_IO)
  find_package(Boost REQUIRED COMPONENTS program_options)
  find_package(GDAL REQUIRED)
  add_library(qesiodeps INTERFACE)
  target_link_libraries(qesiodeps INTERFACE Boost::program_options GDAL::GDAL)
  if(WIN32 OR CMAKE_TOOLCHAIN_FILE)
    find_package(netCDFCxx CONFIG REQUIRED)
    target_link_libraries(qesiodeps INTERFACE netCDF::netcdf-cxx4)
  else()
    set(NETCDF_CXX YES)
    find_package(NetCDF REQUIRED)
    target_include_directories(qesiodeps INTERFACE ${NETCDF_INCLUDES} ${NETCDF_INCLUDES_CXX})
    target_link_libraries(qesiodeps INTERFACE ${NETCDF_LIBRARIES_CXX} ${NETCDF_LIBRARIES})
  endif()
endif()

# Explicit opt-in: adding QES as a subdirectory never probes CUDA.
set(HAS_CUDA_SUPPORT OFF)
set(HAS_OPTIX_SUPPORT OFF)
if(QES_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(HAS_CUDA_SUPPORT ON)
  target_compile_definitions(qesbuild INTERFACE HAS_CUDA BOOST_DISABLE_CURRENT_LOCATION)
  target_link_libraries(qesbuild INTERFACE CUDA::cudart CUDA::curand)
endif()
if(QES_ENABLE_OPTIX)
  if(NOT QES_ENABLE_CUDA)
    message(FATAL_ERROR "QES_ENABLE_OPTIX requires QES_ENABLE_CUDA")
  endif()
  find_package(OptiX 7.0 REQUIRED)
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Optix_macros.cmake")
  set(HAS_OPTIX_SUPPORT ON)
  set(PTX_DIR "${CMAKE_CURRENT_BINARY_DIR}/ptx")
  file(MAKE_DIRECTORY "${PTX_DIR}")
  set(CUDA_GENERATED_OUTPUT_DIR "${PTX_DIR}")
  find_program(BIN2C bin2c REQUIRED)
  target_include_directories(qesbuild INTERFACE ${OptiX_INCLUDE} "${PTX_DIR}")
  target_compile_definitions(qesbuild INTERFACE HAS_OPTIX PTX_DIR="${PTX_DIR}")
endif()

# Compatibility for the standalone modules; all dependencies remain target-scoped.
function(link_external_libraries target)
  target_link_libraries(${target} PRIVATE qesbuild qeswindsio qesutilio)
endfunction()
set(QES_INSTALL_EXECUTABLES "")
add_subdirectory(src/util)
add_subdirectory(src/winds)
if(QES_BUILD_PLUME)
  add_subdirectory(src/plume)
endif()
if(QES_BUILD_FIRE)
  add_subdirectory(src/fire)
endif()
if(QES_BUILD_EXECUTABLES)
  add_subdirectory(qesWinds)
  list(APPEND QES_INSTALL_EXECUTABLES qesWinds)
  if(QES_BUILD_PLUME)
    add_subdirectory(qesPlume)
    add_subdirectory(qes)
    list(APPEND QES_INSTALL_EXECUTABLES qesPlume qes)
  endif()
  if(QES_BUILD_FIRE)
    add_subdirectory(qesFire)
    list(APPEND QES_INSTALL_EXECUTABLES qesFire)
  endif()
endif()
if(PROJECT_IS_TOP_LEVEL AND QES_BUILD_EXAMPLES)
  if(NOT QES_BUILD_PLUME)
    message(FATAL_ERROR "Examples require QES_BUILD_PLUME")
  endif()
  add_subdirectory(examples)
endif()
if(PROJECT_IS_TOP_LEVEL AND ENABLE_TESTS)
  enable_testing()
  include(CTest)
  add_subdirectory(tests)
endif()
if(PROJECT_IS_TOP_LEVEL AND QES_BUILD_DOCUMENTATION)
  find_package(Doxygen REQUIRED)
  configure_file(docs/DoxygenConfig.in "${CMAKE_CURRENT_BINARY_DIR}/docs/Doxyfile" @ONLY)
  add_custom_target(windsdoc COMMAND ${DOXYGEN_EXECUTABLE} "${CMAKE_CURRENT_BINARY_DIR}/docs/Doxyfile")
endif()

# Embedding projects own their installation and packaging.
if(PROJECT_IS_TOP_LEVEL)
  set(CPACK_PACKAGE_NAME "${CMAKE_PROJECT_NAME}")
  set(CPACK_PACKAGE_VENDOR "UtahEFD")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "QES - Quick Environmental Simulation")
  set(CPACK_PACKAGE_DESCRIPTION "The Quick Environmental Simulation (QES) code is a low-computational-cost framework designed to compute high-resolution wind and concentration fields in complex atmospheric-boundary-layer environments. It contains the following modules: QES-Winds, QES-TURB, QES-Plume and QES-Fire.")
  set(CPACK_PACKAGE_VERSION ${QES_VERSION_INFO})
  set(CPACK_PACKAGE_CONTACT "https://qes-documentation.readthedocs.io/en/latest")
  set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/UtahEFD")

  set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
  set(CPACK_RESOURCE_FILE_README "${CMAKE_CURRENT_SOURCE_DIR}/README.md")

  set(CPACK_PACKAGE_INSTALL_DIRECTORY "${CPACK_PACKAGE_NAME}-${CPACK_PACKAGE_VERSION}")

  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/README.md" DESTINATION "${CPACK_PACKAGE_INSTALL_DIRECTORY}")
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" DESTINATION "${CPACK_PACKAGE_INSTALL_DIRECTORY}")
  install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/data" DESTINATION "${CPACK_PACKAGE_INSTALL_DIRECTORY}")
  install(TARGETS ${QES_INSTALL_EXECUTABLES} RUNTIME DESTINATION ${CPACK_PACKAGE_INSTALL_DIRECTORY}/bin)

  if(WIN32)

    # need to include some DLLs for Windows to work correctly: gdal and boost and others
    # - this copies the DLLs from the build location into the bin of the package
    install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qesWinds/" DESTINATION ${CPACK_PACKAGE_INSTALL_DIRECTORY}/bin FILES_MATCHING PATTERN "*.dll")

    set(CPACK_GENERATOR "NSIS")  # Windows Installer
    set(CPACK_NSIS_EXECUTABLES_DIRECTORY bin)
    set(CPACK_NSIS_DISPLAY_NAME "${CPACK_PACKAGE_NAME} ${CPACK_PACKAGE_VERSION}")
    set(CPACK_NSIS_CONTACT "${CPACK_PACKAGE_CONTACT}")
    set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)

  elseif(APPLE)
    set(CPACK_GENERATOR "DragNDrop")  # macOS DMG package
    set(CPACK_DMG_BACKGROUND_IMAGE "${CMAKE_CURRENT_SOURCE_DIR}/data/qesBackground.png")
  elseif(UNIX)
    # unix package gen has not been tested
    set(CPACK_GENERATOR "DEB;RPM")  # Linux: Deb for Debian/Ubuntu, RPM for Fedora/RHEL
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "UtahEFD <s@utah.edu>")
    # set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= 2.28), libstdc++6 (>= 8)")
    set(CPACK_RPM_PACKAGE_LICENSE "GPL-3.0-only")
  endif()

  include( CPack )
endif()
